+function topbarAuditOptions() {
+  const options = [];

+  (state.savedFlows || []).forEach((flow) => options.push({ group: 'Local Saved Audits', value: `local:${flow.id}`, label: flow.name || 'Saved audit' }));
+  if (typeof workbookState !== 'undefined' && workbookState.loaded) {

+    options.push({ group: 'Current Draft', value: state.activeAuditValue, label: state.audit.name || 'Unsaved Draft' });
+  }
+  if (hasSelectedAudit() && state.activeAuditValue && !options.some((option) => option.value === state.activeAuditValue) && state.audit.name) {
+    options.push({ group: 'Current Audit', value: state.activeAuditValue, label: state.audit.name });
+  }

+  const nextPage = AUDIT_FLOW_PAGES[currentIndex + 1];
+  back.hidden = !previous;
+  if (previous) {
+    back.textContent = `Back: ${previous.label}`;
+    back.dataset.flowTarget = previous.page;
+  } else {
+    back.removeAttribute('data-flow-target');
+  }

+  if (nextPage) {
+    next.textContent = `Next: ${current.nextLabel || nextPage.label}`;
+    next.dataset.flowTarget = nextPage.page;

+    created_at AS "createdAt", updated_at AS "updatedAt", updated_by_username AS "updatedByUsername"
+    FROM atlas_audits WHERE archived_at IS NULL ORDER BY lower(name), id`);
+  return result.rows;
+}
+export async function getAudit(pool, auditId) {
+  const result = await pool.query(`SELECT a.*, r.document, r.created_at AS revision_created_at,
+    r.created_by_username AS revision_created_by_username
+    FROM atlas_audits a JOIN atlas_audit_revisions r ON r.id = a.current_revision_id
+    WHERE a.id = $1 AND a.archived_at IS NULL`, [auditId]);
+  if (!result.rowCount) return null;
+  const row = result.rows[0];
+  return { audit: row.document, revision: { id: Number(row.current_revision_id), number: row.current_revision_number,
+    createdAt: row.revision_created_at, createdByUsername: row.revision_created_by_username } };
+}
+async function insertStepRevision(client, { auditId, auditRevisionId, step, operation, user, changeSummary, restoredFromStepRevisionId = null }) {
+  const current = await client.query(`SELECT COALESCE(max(step_revision_number), 0)::integer + 1 AS next
+    FROM atlas_audit_step_revisions WHERE audit_id = $1 AND step_id = $2`, [auditId, step.id]);
+  const [uid, username, fullName] = userFields(user);
+  await client.query(`INSERT INTO atlas_audit_steps
+    (audit_id, step_id, parent_step_id, branch_type, sort_order, step_type, enabled, auto_generated_reference, locked, deleted_at)
+    VALUES ($1,$2,$3,$4,$5,$6,$7,$8,$9,NULL)
+    ON CONFLICT (audit_id, step_id) DO UPDATE SET parent_step_id=EXCLUDED.parent_step_id,
+      branch_type=EXCLUDED.branch_type, sort_order=EXCLUDED.sort_order, step_type=EXCLUDED.step_type,
+      enabled=EXCLUDED.enabled, auto_generated_reference=EXCLUDED.auto_generated_reference,
+      locked=EXCLUDED.locked, deleted_at=NULL`, [auditId, step.id, step.parentId || null, step.branchType || 'root',
+      Number(step.sortOrder || 0), step.step_type || step.type || null, step.enabled !== false,
+      step.autoGeneratedReference === true, step.locked === true]);
+  const inserted = await client.query(`INSERT INTO atlas_audit_step_revisions
+    (audit_id, step_id, audit_revision_id, step_revision_number, operation, step_document,
+     step_sha256, change_summary, created_by_user_id, created_by_username, created_by_full_name, restored_from_step_revision_id)
+    VALUES ($1,$2,$3,$4,$5,$6::jsonb,$7,$8,$9,$10,$11,$12) RETURNING id`, [auditId, step.id, auditRevisionId,
+      current.rows[0].next, operation, JSON.stringify(step), documentHash(step), changeSummary || null,
+      uid, username, fullName, restoredFromStepRevisionId]);
+  await client.query(`UPDATE atlas_audit_steps SET current_step_revision_id=$3 WHERE audit_id=$1 AND step_id=$2`, [auditId, step.id, inserted.rows[0].id]);

+export async function saveAudit(pool, { document, expectedRevisionId, user, changeType = 'save', changeSummary = null, restoredFromRevisionId = null, restoredStep = null }) {
+  document = validateAudit(structuredClone(document));
+  const client = await pool.connect();
+  try {
+    await client.query('BEGIN');
+    const existing = await client.query('SELECT * FROM atlas_audits WHERE id=$1 FOR UPDATE', [document.id]);
+    const isCreate = !existing.rowCount;
+    if (!isCreate && expectedRevisionId != null && Number(existing.rows[0].current_revision_id) !== Number(expectedRevisionId)) {
+      const error = new Error('This audit has a newer database revision. Load the latest revision before saving.');
+      error.status = 409; error.currentRevisionId = Number(existing.rows[0].current_revision_id); throw error;
+    }
+    const hash = documentHash(document);
+    if (!isCreate) {
+      const currentDoc = await client.query('SELECT document_sha256 FROM atlas_audit_revisions WHERE id=$1', [existing.rows[0].current_revision_id]);
+      if (currentDoc.rows[0]?.document_sha256 === hash && changeType === 'save') {
+        await client.query('ROLLBACK');
+        return getAudit(pool, document.id);
+      }
+    }
+    const revisionNumber = isCreate ? 1 : Number(existing.rows[0].current_revision_number) + 1;
+    const [uid, username, fullName, email] = userFields(user);
+    if (isCreate) {
+      await client.query(`INSERT INTO atlas_audits
+        (id,name,purpose,description,schema_version,source_kind,owner_user_id,owner_username,
+         created_by_user_id,created_by_username,updated_by_user_id,updated_by_username)
+        VALUES ($1,$2,$3,$4,$5,$6,$7,$8,$7,$8,$7,$8)`, [document.id, document.name,
+          document.purpose || '', document.description || '', Number(document.schemaVersion || 1),
+          document.sourceKind || 'database', uid, username]);
+    }
+    const revision = await client.query(`INSERT INTO atlas_audit_revisions
+      (audit_id,revision_number,parent_revision_id,document,document_sha256,change_type,change_summary,
+       created_by_user_id,created_by_username,created_by_full_name,created_by_email,restored_from_revision_id)
+      VALUES ($1,$2,$3,$4::jsonb,$5,$6,$7,$8,$9,$10,$11,$12) RETURNING id, created_at`, [document.id,
+        revisionNumber, isCreate ? null : existing.rows[0].current_revision_id, JSON.stringify(document), hash,
+        isCreate ? 'create' : changeType, changeSummary, uid, username, fullName, email, restoredFromRevisionId]);
+    const auditRevisionId = Number(revision.rows[0].id);
+    let previousSteps = new Map();
+    if (!isCreate) {
+      const previous = await client.query('SELECT document FROM atlas_audit_revisions WHERE id=$1', [existing.rows[0].current_revision_id]);
+      previousSteps = new Map((previous.rows[0]?.document?.steps || []).map(step => [step.id, step]));
+    }
+    const submitted = new Map(document.steps.map(step => [step.id, step]));
+    for (const step of document.steps) {
+      const before = previousSteps.get(step.id);
+      let operation = before ? 'update' : 'create';
+      if (before && documentHash(before) === documentHash(step)) continue;
+      if (before && (before.parentId !== step.parentId || Number(before.sortOrder || 0) !== Number(step.sortOrder || 0))) operation = 'move';
+      if (restoredStep?.stepId === step.id) operation = 'restore';
+      await insertStepRevision(client, { auditId: document.id, auditRevisionId, step, operation, user, changeSummary,
+        restoredFromStepRevisionId: restoredStep?.stepId === step.id ? restoredStep.stepRevisionId : null });
+    }
+    for (const [stepId, step] of previousSteps) if (!submitted.has(stepId)) {
+      const current = await client.query(`SELECT COALESCE(max(step_revision_number),0)::integer+1 AS next FROM atlas_audit_step_revisions WHERE audit_id=$1 AND step_id=$2`, [document.id, stepId]);
+      await client.query(`INSERT INTO atlas_audit_step_revisions
+        (audit_id,step_id,audit_revision_id,step_revision_number,operation,step_document,step_sha256,change_summary,created_by_user_id,created_by_username,created_by_full_name)
+        VALUES ($1,$2,$3,$4,'delete',$5::jsonb,$6,$7,$8,$9,$10)`, [document.id, stepId, auditRevisionId, current.rows[0].next,
+          JSON.stringify(step), documentHash(step), changeSummary, uid, username, fullName]);
+      await client.query('UPDATE atlas_audit_steps SET deleted_at=current_timestamp,current_step_revision_id=NULL WHERE audit_id=$1 AND step_id=$2', [document.id, stepId]);
+    }
+    await client.query(`UPDATE atlas_audits SET name=$2,purpose=$3,description=$4,schema_version=$5,source_kind=$6,
+      current_revision_id=$7,current_revision_number=$8,updated_by_user_id=$9,updated_by_username=$10,updated_at=current_timestamp
+      WHERE id=$1`, [document.id, document.name, document.purpose || '', document.description || '',
+      Number(document.schemaVersion || 1), document.sourceKind || 'database', auditRevisionId, revisionNumber, uid, username]);
+    await client.query('COMMIT');
+    return { audit: document, revision: { id: auditRevisionId, number: revisionNumber,
+      createdAt: revision.rows[0].created_at, createdByUsername: username } };
+  } catch (error) { await client.query('ROLLBACK'); throw error; } finally { client.release(); }
+}

+    step_document AS "stepDocument",step_sha256 AS sha256,created_by_username AS "createdByUsername",created_at AS "createdAt"

+export async function restoreStepRevision(pool, { auditId, stepId, revisionId, expectedRevisionId, user }) {
+  const [current, stepRevision] = await Promise.all([getAudit(pool, auditId), getStepRevision(pool, auditId, stepId, revisionId)]);
+  if (!current || !stepRevision?.stepDocument) return null;
+  const document = structuredClone(current.audit);
+  const index = document.steps.findIndex(step => step.id === stepId);
+  if (index >= 0) document.steps[index] = stepRevision.stepDocument; else document.steps.push(stepRevision.stepDocument);
+  return saveAudit(pool, { document, expectedRevisionId, user, changeType: 'restore',
+    changeSummary: `Restored step ${stepId} revision ${stepRevision.number}`,
+    restoredStep: { stepId, stepRevisionId: Number(revisionId) } });

+  if (!result.rowCount) { const error = new Error('This audit has changed or is already archived.'); error.status=409; throw error; }
+  return { id: auditId, archived: true };
